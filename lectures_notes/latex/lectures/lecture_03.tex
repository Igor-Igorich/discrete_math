\subsection{Основные понятия комбинаторики}

\begin{definition}
    \textdef{Комбинаторика} --- это раздел дискретной математики, изучающий 
    дискретные объекты, конечные множества (\textdef{комбинаторные 
    конфигурации}) и отношения на них. Важнейшей задачей перечислительной 
    комбинаторики является подсчёт числа элементов конечных множеств, 
    удовлетворяющих заданным условиям.
\end{definition}

\begin{definition}
    \textdef{Комбинаторные конфигурации} --- это структурированные 
    определённым образом наборы объектов, составленные из элементов 
    конечных множеств по заданным правилам.
\end{definition}

\begin{example}
    Классическими примерами комбинаторных конфигураций являются 
    перестановки, размещения, сочетания и разбиения.
\end{example}

\begin{definition}[Классификация наборов]
    Любой конечный \textdef{набор} объектов в комбинаторике классифицируется 
    по двум независимым критериям:
    \begin{itemize}
        \item \textbf{По упорядоченности:}
        \begin{itemize}
            \item \emph{Неупорядоченный набор}: порядок следования элементов 
            не имеет значения.
            \item \emph{Упорядоченный набор}: порядок элементов строго 
            зафиксирован (\textdef{кортеж}).
        \end{itemize}
        \item \textbf{По допустимости повторений:}
        \begin{itemize}
            \item \emph{Без повторений}: все элементы в наборе попарно 
            различны. Неупорядоченный набор без повторений образует 
            классическое \textdef{множество}.
            \item \emph{С повторениями}: элементы могут входить в набор 
            несколько раз (с некоторой кратностью). Неупорядоченный набор с 
            повторениями называется \textdef{мультимножеством}.
        \end{itemize}
    \end{itemize}
\end{definition}

\begin{designation}
    Для краткого указания количества элементов $n$ используются 
    термины:
    \begin{itemize}
        \item \textdef{$n$-набор} --- набор, состоящий из $n$ элементов;
        \item \textdef{$n$-множество} --- множество мощности $n$;
        \item \textdef{$n$-мультимножество} --- мультимножество мощности $n$ 
        (с учётом кратностей элементов);
        \item \textdef{$n$-кортеж} --- упорядоченный набор длины $n$.
    \end{itemize}
\end{designation}

\begin{definition}
    \textdef{$(n, k)$-выборкой} называется любой $k$-набор, составленный из 
    элементов заданного $n$-множества.
\end{definition}

\subsection{Базовые принципы комбинаторики}

В данном разделе все рассматриваемые множества предполагаются конечными.

\begin{theorem}[\textbf{0. Принцип биективности}]
    Если между двумя конечными множествами $X$ и $Y$ существует биективное 
    отображение, то они содержат одинаковое количество элементов: 
    $\abs{X} = \abs{Y}$.
\end{theorem}

\begin{theorem}[\textbf{1. Принцип сложения}]
    Если конечные множества $X_1$ и $X_2$ не пересекаются 
    ($X_1 \cap X_2 = \varnothing$), то мощность их объединения равна сумме 
    их мощностей:
    \begin{equation} \label{eq:sum_rule}
        \abs{X_1 \cup X_2} = \abs{X_1} + \abs{X_2} \tag{1}
    \end{equation}
\end{theorem}

\begin{theorem}[\textbf{2. Принцип умножения}]
    Мощность декартова произведения конечных множеств равна произведению 
    их мощностей:
    \begin{equation} \label{eq:prod_rule}
        \abs{X_1 \times X_2 \times \dots \times X_m} = 
        \abs{X_1} \cdot \abs{X_2} \cdot \dots \cdot \abs{X_m} \tag{2}
    \end{equation}
\end{theorem}

\begin{example}
    Пусть из города $A$ в город $B$ ведут $3$ дороги, а из города $B$ в 
    город $C$ --- $2$ дороги. Обозначим множество дорог на первом участке 
    через $X_1$ ($\abs{X_1} = 3$), а на втором --- через $X_2$ 
    ($\abs{X_2} = 2$). Любой составной маршрут из $A$ в $C$ через $B$ 
    представляет собой упорядоченную пару из декартова произведения 
    $X_1 \times X_2$. По принципу умножения \eqref{eq:prod_rule} общее 
    количество маршрутов равно $3 \cdot 2 = 6$.
\end{example}

\begin{theorem}[\textbf{3. Принцип Дирихле}]
    Пусть $m$ предметов («кроликов») распределены по $n$ классам 
    («ящикам»).
    \begin{itemize}
        \item \emph{Классическая формулировка:} Если $m > n$, то хотя бы в 
        одном ящике окажется не менее двух предметов ($\ge 2$).
        \item \emph{Обобщённая формулировка:} Если $m > n \cdot k$ 
        (где $k \in \N \cup \{0\}$), то хотя бы в одном ящике окажется 
        строго больше $k$ предметов (то есть не менее $k + 1$ предмета).
    \end{itemize}
\end{theorem}

\begin{proof}[Доказательство обобщённого принципа Дирихле]~\\
    \begin{quote}
        Докажем утверждение от противного. Пусть $m > n \cdot k$ предметов 
        распределены по $n$ ящикам, но ни в одном ящике нет более $k$ 
        предметов (то есть в $i$-м ящике находится $m_i \le k$ предметов). 
        Тогда общее число предметов во всех ящиках удовлетворяет оценке:
        \[
            m = \sum_{i=1}^n m_i \le \sum_{i=1}^n k = n \cdot k
        \]
        Это противоречит условию $m > n \cdot k$. Следовательно, хотя бы в 
        одном ящике находится не менее $k + 1$ предмета.
    \end{quote}
\end{proof}

\begin{example} ~
    В квадрат со стороной $1$ произвольным образом бросили $5$ точек. 
    Докажем, что среди них обязательно найдутся хотя бы $2$ точки, 
    расстояние между которыми не превышает $1/\sqrt{2}$.

    \begin{proof}[Решение]~\\
        \begin{quote}
            Разобьём исходный квадрат $1 \times 1$ двумя средними линиями на 
            $n = 4$ равных малых квадрата размером $1/2 \times 1/2$. 
            В терминах принципа Дирихле точки играют роль «кроликов» 
            ($m = 5$), а малые квадраты --- роль «ящиков» ($n = 4$).

            Так как $5 > 4$, по принципу Дирихле хотя бы в один из малых 
            квадратов попадут не менее двух точек. Максимальное расстояние 
            между двумя точками внутри квадрата со стороной $1/2$ равно

            \]
            Следовательно, расстояние между этими двумя точками не

        \end{quote}
    \end{proof}

\begin{theorem}[\textbf{4. Принцип включений и исключений}]
    Для двух конечных множеств мощность их объединения выражается формулой:
    \begin{equation} \label{eq:pie_2}
        \abs{X_1 \cup X_2} = \abs{X_1} + \abs{X_2} - \abs{X_1 \cap X_2} \tag{3}
    \end{equation}
    Для трёх множеств:
    \begin{equation} \label{eq:pie_3}
        \begin{aligned}
            \abs{X_1 \cup X_2 \cup X_3} &= \abs{X_1} + \abs{X_2} + \abs{X_3} 
            - \abs{X_1 \cap X_2} - \abs{X_1 \cap X_3} - \abs{X_2 \cap X_3} \\
            &\quad + \abs{X_1 \cap X_2 \cap X_3}
        \end{aligned} \tag{4}
    \end{equation}
    В общем случае для $n$ конечных множеств справедлива формула:
    \begin{equation} \label{eq:pie_n}
        \abs{X_1 \cup X_2 \cup \dots \cup X_n} = 
        \sum_{k=1}^{n} (-1)^{k-1} 
        \left( \sum_{1 \le i_1 < i_2 < \dots < i_k \le n} 
        \abs{X_{i_1} \cap \dots \cap X_{i_k}} \right) \tag{5}
    \end{equation}
\end{theorem}

\begin{proof}[Доказательство принципа включений и исключений]~\\
    \begin{quote}
        Покажем, что любой элемент $x \in X_1 \cup \dots \cup X_n$ 
        учитывается в правой части формулы \eqref{eq:pie_n} ровно $1$ раз.

        Пусть фиксированный элемент $x$ принадлежит ровно $m$ множествам из 
        семейства $X_1, \dots, X_n$ ($1 \le m \le n$). В левой части 
        равенства \eqref{eq:pie_n} он даёт вклад, равный $1$. Подсчитаем 
        его суммарный вклад в правую часть:
        \begin{itemize}
            \item При $k = 1$ (сумма мощностей отдельных множеств) элемент 
            $x$ учитывается $\binom{m}{1}$ раз со знаком «$+$».
            \item При $k = 2$ (сумма попарных пересечений) элемент $x$ 
            входит ровно в те пересечения $X_{i_1} \cap X_{i_2}$, оба 
            множества которых содержат $x$, то есть учитывается 
            $\binom{m}{2}$ раз со знаком «$-$».
            \item В общем случае при произвольном $k \le m$ в сумме 
            $k$-кратных пересечений элемент $x$ учитывается $\binom{m}{k}$ 
            раз со знаком $(-1)^{k-1}$ (при $k > m$ вклад равен $0$).
        \end{itemize}
        Таким образом, суммарный вклад элемента $x$ в правую часть равен:
        \[
            S = \binom{m}{1} - \binom{m}{2} + \binom{m}{3} - \dots + 
            (-1)^{m-1} \binom{m}{m}
        \]
        По формуле бинома Ньютона разложим выражение $(1 - 1)^m$:
        \[
            0 = (1 - 1)^m = \binom{m}{0} - \binom{m}{1} + \binom{m}{2} - 
            \dots + (-1)^m \binom{m}{m}
        \]
        Учитывая, что $\binom{m}{0} = 1$, перенесём все остальные слагаемые 
        в левую часть:
        \[
            \binom{m}{1} - \binom{m}{2} + \dots + (-1)^{m-1} \binom{m}{m} = 1
        \]
        Следовательно, $S = 1$, и каждый элемент объединения учитывается 
        формулой \eqref{eq:pie_n} ровно один раз.
    \end{quote}
\end{proof}

\subsubsection{Частные случаи применения принципов}

\begin{proposition}[\textbf{Частный случай №1: слова в алфавите}]
    Пусть $\Sigma$ --- конечный алфавит. Количество всех слов длины $k$ над 
    алфавитом $\Sigma$ равно:
    \begin{equation} \label{eq:words_count}
        \abs{\Sigma^k} = \abs{\Sigma}^k \tag{6}
    \end{equation}
    В частности, количество двоичных слов длины $k$ над булевым алфавитом 
    $\B = \{0, 1\}$ равно $\abs{\B^k} = 2^k$.
\end{proposition}

\begin{proof}[Обоснование]~\\
    \begin{quote}
        Множество слов длины $k$ есть декартова степень $\Sigma^k = 
        \Sigma \times \dots \times \Sigma$ ($k$ раз). Применяя принцип 
        умножения \eqref{eq:prod_rule}, получаем 
        $\abs{\Sigma^k} = \abs{\Sigma}^k$.
    \end{quote}
\end{proof}

\begin{proposition}[\textbf{Частный случай №2: булеан конечного множества}]
    Для любого конечного множества $X = \{x_1, \dots, x_n\}$ мощность его 
    булеана равна:
    \[
        \abs{\mathcal{P}(X)} = 2^{\abs{X}} = 2^n
    \]
\end{proposition}

\begin{proof}[Доказательство через характеристические векторы]~\\
    \begin{quote}
        Построим биекцию между множеством подмножеств $\mathcal{P}(X)$ и 
        множеством бинарных векторов длины $n$ ($\B^n$). Каждому 
        подмножеству $Y \subseteq X$ сопоставим его характеристический 
        бинарный вектор $(b_1, b_2, \dots, b_n) \in \B^n$, где $b_i = 1$, 
        если $x_i \in Y$, и $b_i = 0$, если $x_i \notin Y$.

        Например, при $n = 5$ подмножеству $\{x_1, x_4, x_5\}$ однозначно 
        соответствует вектор $10011$. По принципу биективности и формуле 
        \eqref{eq:words_count} получаем $\abs{\mathcal{P}(X)} = \abs{\B^n} = 2^n$.

\subsection{Перестановки, размещения и сочетания}

\subsubsection{Перестановки}

\begin{definition}
    \textdef{Перестановкой} конечного множества $X = \{x_1, \dots, x_n\}$ 
    называется любое биективное отображение множества $X$ в само себя 
    ($X \to X$), или, эквивалентно, упорядоченный $n$-набор без повторений 
    из элементов $X$.
\end{definition}

\begin{designation}
    Перестановку как биекцию записывают в виде двухстрочной таблицы:
    \[
        \begin{pmatrix} 
            x_1 & x_2 & \dots & x_n \\ 
            x_{i_1} & x_{i_2} & \dots & x_{i_n} 
        \end{pmatrix}
    \]
    При фиксированном порядке исходных элементов перестановку записывают 
    одной строкой (кортежем): $(x_{i_1}, x_{i_2}, \dots, x_{i_n})$.
\end{designation}

\begin{proposition}[\textbf{Частный случай №3: число перестановок}]
    Число всех перестановок $n$-элементного множества равно $P_n = n!$.
\end{proposition}

\begin{theorem}[\textbf{Формула Стирлинга}]
    При $n \to \infty$ для факториала справедлива асимптотическая формула:
    \begin{equation} \label{eq:stirling}
        n! \sim \sqrt{2 \pi n} \left(\frac{n}{e}\right)^n \tag{7}
    \end{equation}
\end{theorem}

\begin{proof}[Эскиз вывода формулы Стирлинга]~\\
    \begin{quote}
        Прологарифмируем факториал, заменив логарифм произведения суммой 
        логарифмов:
        \[
            \ln(n!) = \ln(1 \cdot 2 \cdot \dots \cdot n) = \sum_{k=1}^{n} \ln k
        \]
        Приблизим полученную интегральную сумму определённым интегралом от 
        монотонно возрастающей функции $\ln x$:
        \[
            \sum_{k=1}^{n} \ln k \approx \int_{1}^{n} \ln x \, dx
        \]
        Интегрируя по частям (первообразная равна $x \ln x - x$), по 
        формуле Ньютона---Лейбница находим:
        \[
            \int_{1}^{n} \ln x \, dx = (x \ln x - x)\Big|_{1}^{n} = 
            n \ln n - n + 1
        \]
        Отбрасывая константу при больших $n$, имеем $\ln(n!) \approx 
        n \ln n - n$. Потенцируя, получаем главный показатель роста:
        \[
            n! \approx e^{n \ln n - n} = (e^{\ln n})^n \cdot e^{-n} = 
            \left(\frac{n}{e}\right)^n
        \]
        Учёт погрешности метода трапеций (формула Эйлера---Маклорена) и 
        вычисление константы через формулу Валлиса дают предэкспоненциальный 
        множитель $\sqrt{2 \pi n}$, приводя к асимптотике \eqref{eq:stirling}.
    \end{quote}
\end{proof}

\subsubsection{Размещения (Arrangements)}

\begin{definition}
    \textdef{Размещением без повторений} из $n$ элементов по $k$ называется 
    любая упорядоченная $(n, k)$-выборка без повторений (обозначается 
    $A_n^k$ или $(n)_k$).
\end{definition}

\begin{proposition}
    Число размещений без повторений из $n$ по $k$ ($k \le n$) вычисляется 
    по формуле:
    \begin{equation} \label{eq:arr_no_rep}
        A_n^k = n \cdot (n - 1) \cdot (n - 2) \cdot \dots \cdot (n - k + 1) = 
        \frac{n!}{(n - k)!} \tag{8}
    \end{equation}
\end{proposition}

\begin{definition}
    \textdef{Размещением с повторениями} из $n$ элементов по $k$ называется 
    любая упорядоченная $(n, k)$-выборка с возможными повторениями 
    элементов (обозначается $\bar{A}_n^k$).
\end{definition}

\begin{proposition}
    Число размещений с повторениями из $n$ по $k$ равно:
    \begin{equation} \label{eq:arr_rep}
        \bar{A}_n^k = n^k \tag{9}
    \end{equation}
\end{proposition}

\begin{proof}[Обоснование формул для размещений]~\\
    \begin{quote}
        Для размещений без повторений на первую позицию кортежа можно 
        выбрать любой из $n$ элементов, на вторую --- любой из $n - 1$ 
        оставшихся, и так далее до $k$-й позиции, на которую остаётся 
        $n - k + 1$ вариантов. Применяя принцип умножения \eqref{eq:prod_rule}, 
        получаем \eqref{eq:arr_no_rep}.

        Для размещений с повторениями выбор на каждом из $k$ шагов 
        независим и допускает любой из $n$ элементов, что по принципу 
        умножения даёт $n^k$.
    \end{quote}
\end{proof}

\subsubsection{Сочетания (Combinations)}

\begin{definition}
    \textdef{Сочетанием без повторений} из $n$ элементов по $k$ называется 
    любая неупорядоченная $(n, k)$-выборка без повторений, то есть 
    $k$-элементное подмножество $n$-множества (обозначается $C_n^k$ или 
    $\binom{n}{k}$).
\end{definition}

\begin{proposition}
    Число сочетаний без повторений из $n$ по $k$ ($0 \le k \le n$) равно:
    \begin{equation} \label{eq:comb_no_rep}
        C_n^k = \binom{n}{k} = \frac{A_n^k}{k!} = \frac{n!}{k!(n - k)!} \tag{10}
    \end{equation}
\end{proposition}

\begin{proof}[Доказательство через факторизацию по отношению эквивалентности]~\\
    \begin{quote}
        Пусть $X$ --- множество всех размещений без повторений ($k$-кортежей) 
        из $n$-множества, причём $\abs{X} = A_n^k$. Введём на множестве $X$ 
        отношение эквивалентности $\sim$: два $k$-кортежа эквивалентны 
        тогда и только тогда, когда они состоят из одного и того же набора 
        элементов и отличаются лишь порядком их следования.

        Каждый класс эквивалентности однозначно соответствует одному 
        неупорядоченному $k$-подмножеству (сочетанию), поэтому число 
        сочетаний $C_n^k$ равно мощности фактор-множества $\abs{X/{\sim}}$. 
        Поскольку $k$ различных элементов внутри фиксированного набора 
        можно переставить ровно $k!$ способами, каждый класс 
        эквивалентности содержит ровно $k!$ кортежей. Так как классы не 
        пересекаются, получаем $\abs{X/{\sim}} = \abs{X} / k! = A_n^k / k!$.
    \end{quote}
\end{proof}

\begin{definition}
    \textdef{Сочетанием с повторениями} из $n$ элементов по $k$ называется 
    любая неупорядоченная $(n, k)$-выборка с возможными повторениями, то 
    есть $k$-мультимножество, составленное из элементов $n$-множества 
    (обозначается $\bar{C}_n^k$).
\end{definition}

\begin{proposition}
    Число сочетаний с повторениями из $n$ по $k$ вычисляется по формуле:
    \begin{equation} \label{eq:comb_rep}
        \bar{C}_n^k = C_{n+k-1}^{n-1} = C_{n+k-1}^k = 
        \frac{(n + k - 1)!}{k!(n - 1)!} \tag{11}
    \end{equation}
\end{proposition}

\begin{proof}[Доказательство методом шаров и перегородок]~\\
    \begin{quote}
        Задача выбора неупорядоченного набора из $k$ элементов (с 
        повторениями) среди $n$ доступных типов равносильна разложению $k$ 
        неразличимых шаров по $n$ различимым ящикам (где номер ящика 
        соответствует типу элемента, а количество шаров в ящике --- 
        кратности вхождения этого элемента в мультимножество).

        Выстроим $k$ шаров в ряд и разделим их на $n$ групп с помощью 
        $n - 1$ межъящичных перегородок. Каждое такое распределение 
        кодируется словом длины $n + k - 1$ в алфавите $\{0, 1\}$, где 
        символ «$0$» обозначает шар ($k$ штук), а символ «$1$» --- 
        перегородку ($n - 1$ штук).

        Например, для $n = 4$ ящиков ($3$ перегородки) и $k = 4$ шаров 
        бинарный вектор $0110010$ означает: в $1$-м ящике лежит $1$ шар, 
        во $2$-м --- $0$ шаров, в $3$-м --- $2$ шара, в $4$-м --- $1$ шар.

        \begin{figure}[H]
            \centering
            \begin{tikzpicture}[scale=0.75]

                \draw[thick] (0, 0) rectangle (1, 1) node[midway] {$0$};

        Построенное соответствие биективно. Чтобы задать такой вектор длины 
        $n + k - 1$, достаточно выбрать $n - 1$ позицию для единиц (или,

        $C_{n+k-1}^{n-1} = C_{n+k-1}^k$ способами.
    \end{quote}
\end{proof}

\subsection{Пример решения задачи}

\begin{problem}
    Дано натуральное число $720$. Сколько всего натуральных (целых 
    положительных) делителей имеет это число?
\end{problem}

\begin{proof}[Решение]~\\
    \begin{quote}
        Разложим число $720$ на простые множители (каноническое разложение 
        по основной теореме арифметики):
        \[
            720 = 2^4 \cdot 3^2 \cdot 5^1
        \]
        Любой натуральный делитель $d$ числа $720$ однозначно записывается 
        в виде:
        \[
            d = 2^a \cdot 3^b \cdot 5^c
        \]
        где показатели степеней $a, b, c$ выбираются независимо из 
        множеств:
        \begin{itemize}
            \item $a \in \{0, 1, 2, 3, 4\}$ --- всего $5$ вариантов;
            \item $b \in \{0, 1, 2\}$ --- всего $3$ варианта;
            \item $c \in \{0, 1\}$ --- всего $2$ варианта.
        \end{itemize}
        Таким образом, каждый делитель взаимно однозначно соответствует 
        тройке показателей $(a, b, c)$. По принципу умножения 
        \eqref{eq:prod_rule} общее количество натуральных делителей равно:
        \[
            5 \cdot 3 \cdot 2 = 30
        \]
        \textbf{Ответ:} $30$ натуральных делителей.
    \end{quote}
\end{proof}
